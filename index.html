async function sendMessage() { const text = inputEl.value.trim(); const imageToSend = pendingImage; if (!text && !imageToSend) return; // Grab current chat history before appending the new message const historyToSend = [...currentConversation.messages]; renderMessageToDOM('user', text, imageToSend); currentConversation.messages.push({ sender: 'user', text, image: imageToSend }); persistCurrentConversation(); inputEl.value = ""; clearPendingImage(); sendBtn.disabled = true; const typingBubble = addTyping(); try { const body = { message: text, history: historyToSend }; if (imageToSend) body.image = imageToSend; const res = await fetch(BACKEND_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body), }); const data = await res.json(); const reply = data.reply || "Sorry, something went wrong."; const textWrap = document.createElement('div'); renderMathContent(textWrap, reply); typingBubble.innerHTML = ''; typingBubble.appendChild(textWrap); scrollToBottom(); currentConversation.messages.push({ sender: 'bot', text: reply }); persistCurrentConversation(); } catch (err) { typingBubble.textContent = "Could not reach the chatbot backend. Is it running?"; } finally { sendBtn.disabled = false; } }
